\chapter{Glossary of Terms Used in This Report}
\label{ch:glossary-of-terms-used-in-this-report}
\begin{longtable}{|p{0.42\textwidth}|p{0.42\textwidth}|}
\hline
\rowcolor{thesisgreen}\textcolor{white}{\textbf{Term}} & \textcolor{white}{\textbf{Meaning}} \\
\hline\endhead
DAM & Delegation of Authority Matrix — the source document this project makes queryable \\
\hline
I / C / R / A & Initiate / Check-verify-or-consult / Review / Approve — the DAM's authority-code notation \\
\hline
(i) & The DAM's informed-party marker (a role that must simply be told about an activity) \\
\hline
Node & One structured record in this project's data model (a chapter, process, task, child task, or threshold variant) \\
\hline
Grounding check & The verification step ensuring an LLM's rephrased answer still contains every real fact, verbatim \\
\hline
RAG & Retrieval-Augmented Generation — here, deterministic retrieval feeding a fact-constrained LLM rephrasing step \\
\hline
TF-IDF & Term Frequency–Inverse Document Frequency, the lexical text-matching method used for free-text search \\
\hline
\texttt{MultiDiGraph} & A directed graph type (networkx) permitting multiple parallel edges between the same two nodes \\
\hline
Threshold variant & A DAM sub-item keyed by a monetary threshold band rather than a sequential number \\
\hline
OOM & Out of memory — the cause of the first failed Render deployment \\
\hline
\end{longtable}
